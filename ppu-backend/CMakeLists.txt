cmake_minimum_required(VERSION 3.18)
project(ggml-ppu-plugin LANGUAGES CXX)

set(LLAMA_CPP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." CACHE PATH "Unmodified llama.cpp source")
set(LLAMA_CPP_BUILD_DIR "${LLAMA_CPP_SOURCE_DIR}/build-host" CACHE PATH "Matching shared llama.cpp build")
set(PPU_SDK "$ENV{PPU_SDK}" CACHE PATH "PPU SDK root")
option(PPU_BACKEND_HOST_TEST "Use a host-only test runtime, not PPU hardware" OFF)

find_path(GGML_INCLUDE_DIR ggml.h PATHS "${LLAMA_CPP_SOURCE_DIR}/ggml/include" NO_DEFAULT_PATH REQUIRED)
find_path(GGML_IMPL_DIR ggml-backend-impl.h PATHS "${LLAMA_CPP_SOURCE_DIR}/ggml/src" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIBRARY ggml-base PATHS "${LLAMA_CPP_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIBRARY ggml PATHS "${LLAMA_CPP_BUILD_DIR}/bin" NO_DEFAULT_PATH REQUIRED)

add_library(ggml-ppu MODULE ggml-ppu.cpp)
target_compile_features(ggml-ppu PRIVATE cxx_std_17)
target_include_directories(ggml-ppu PRIVATE "${GGML_INCLUDE_DIR}" "${GGML_IMPL_DIR}")
target_compile_definitions(ggml-ppu PRIVATE GGML_BACKEND_DL GGML_BACKEND_SHARED GGML_BACKEND_BUILD)
target_link_libraries(ggml-ppu PRIVATE "${GGML_BASE_LIBRARY}")
set_target_properties(ggml-ppu PROPERTIES CXX_VISIBILITY_PRESET hidden CUDA_VISIBILITY_PRESET hidden)

if(PPU_BACKEND_HOST_TEST)
    target_sources(ggml-ppu PRIVATE host-test-runtime.cpp)
    set_target_properties(ggml-ppu PROPERTIES OUTPUT_NAME ggml-ppu-test)
    set(PPU_DEMO_DEVICE PPU-TEST0)
else()
    if(NOT EXISTS "${PPU_SDK}/CUDA_SDK/bin/nvcc")
        message(FATAL_ERROR "Set PPU_SDK to a PPU SDK with CUDA_SDK/bin/nvcc")
    endif()
    set(CMAKE_CUDA_COMPILER "${PPU_SDK}/CUDA_SDK/bin/nvcc" CACHE FILEPATH "PPU CUDA compiler")
    set(CUDAToolkit_ROOT "${PPU_SDK}/CUDA_SDK")
    set(CMAKE_CUDA_ARCHITECTURES OFF CACHE STRING "Let the PPU SDK select its target")
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_sources(ggml-ppu PRIVATE runtime.cu)
    set_target_properties(ggml-ppu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES CUDA_RUNTIME_LIBRARY Shared CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
    target_link_libraries(ggml-ppu PRIVATE CUDA::cudart)
    set(PPU_DEMO_DEVICE PPU0)
endif()

add_executable(ppu-rms-norm-demo demo.cpp)
target_compile_features(ppu-rms-norm-demo PRIVATE cxx_std_17)
target_include_directories(ppu-rms-norm-demo PRIVATE "${GGML_INCLUDE_DIR}")
target_compile_definitions(ppu-rms-norm-demo PRIVATE GGML_RUNTIME_DIR="${LLAMA_CPP_BUILD_DIR}/bin")
target_link_libraries(ppu-rms-norm-demo PRIVATE "${GGML_LIBRARY}" "${GGML_BASE_LIBRARY}")

include(CTest)
add_test(NAME ppu-rms-norm COMMAND ppu-rms-norm-demo "${PPU_DEMO_DEVICE}")
set_tests_properties(ppu-rms-norm PROPERTIES ENVIRONMENT "GGML_BACKEND_PATH=$<TARGET_FILE:ggml-ppu>")
